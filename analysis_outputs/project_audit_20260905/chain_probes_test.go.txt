package chain

import (
    "blockEmulator/core"
    "blockEmulator/params"
    "bytes"
    "math/big"
    "testing"
    "time"
    "github.com/ethereum/go-ethereum/core/rawdb"
)

func auditChain(t *testing.T, node uint64) *BlockChain {
    t.Helper()
    params.ShardNum = 1
    c, err := NewBlockChain(&params.ChainConfig{ShardID: 0, NodeID: node, ShardNums: 1, BlockSize: 10}, rawdb.NewMemoryDatabase())
    if err != nil { t.Fatal(err) }
    t.Cleanup(c.CloseBlockChain)
    return c
}

func TestAuditFollowerDoesNotReproposeCommittedTransaction(t *testing.T) {
    leader := auditChain(t, 10)
    follower := auditChain(t, 11)
    tx := core.NewTransaction("00000000001", "00000000002", big.NewInt(1), 1, time.Now())
    leader.Txpool.AddTx2Pool(tx)
    follower.Txpool.AddTx2Pool(tx)
    b := leader.GenerateBlock(0)
    leader.AddBlock(b)
    follower.AddBlock(b)
    next := follower.GenerateBlock(1)
    for _, repeated := range next.Body {
        if bytes.Equal(repeated.TxHash, tx.TxHash) {
            t.Fatal("follower reproposed an already committed transaction after becoming leader")
        }
    }
}

func TestAuditValidationRejectsWrongStateRoot(t *testing.T) {
    c := auditChain(t, 12)
    b := c.GenerateBlock(0)
    b.Header.StateRoot = bytes.Repeat([]byte{0xee}, 32)
    b.Hash = b.Header.Hash()
    if err := c.IsValidBlock(b); err == nil {
        t.Fatal("IsValidBlock accepted arbitrary state root with correct parent and tx root")
    }
}
